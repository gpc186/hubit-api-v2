// ==========================================
// ENUMS
// ==========================================
enum UserRole {
  CANDIDATE
  COMPANY
  ADMIN
}

enum VerificationStatus {
  PENDING
  VERIFIED
  REJECTED
}

enum JobStatus {
  DRAFT
  PUBLISHED
  PAUSED
  CLOSED
}

enum ApplicationStatus {
  PENDING
  REVIEWING
  INTERVIEW
  APPROVED
  REJECTED
}

enum EvidenceSource {
  GITHUB
  CODEWARS
  MANUAL
}

// ==========================================
// ENTIDADES CORE & AUTENTICAÇÃO
// ==========================================
model User {
  id            String            @id @default(uuid())
  email         String            @unique
  passwordHash  String
  role          UserRole          @default(CANDIDATE)
  createdAt     DateTime          @default(now())
  updatedAt     DateTime          @updatedAt

  refreshTokens RefreshToken[]
  candidate     CandidateProfile?
  company       CompanyProfile?

  @@map("users")
}

model RefreshToken {
  id        String   @id @default(uuid())
  tokenHash String   @unique
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  expiresAt DateTime
  createdAt DateTime @default(now())

  @@map("refresh_tokens")
}

// ==========================================
// PERFIS PROFISSIONAIS & EMPRESARIAIS
// ==========================================
model CandidateProfile {
  id        String   @id @default(uuid())
  userId    String   @unique
  fullName  String
  headline  String?
  bio       String?
  location  String?
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  user         User             @relation(fields: [userId], references: [id], onDelete: Cascade)
  resumes      Resume[]
  applications Application[]
  evidences    Evidence[]
  skills       CandidateSkill[]

  @@map("candidate_profiles")
}

model CompanyProfile {
  id          String             @id @default(uuid())
  userId      String             @unique
  companyName String
  cnpj        String             @unique
  status      VerificationStatus @default(PENDING)
  createdAt   DateTime           @default(now())
  updatedAt   DateTime           @updatedAt

  user          User                  @relation(fields: [userId], references: [id], onDelete: Cascade)
  jobs          Job[]
  verifications CompanyVerification[]

  @@map("company_profiles")
}

model CompanyVerification {
  id         String             @id @default(uuid())
  companyId  String
  status     VerificationStatus
  reviewerId String?
  reason     String?
  createdAt  DateTime           @default(now())

  company CompanyProfile @relation(fields: [companyId], references: [id], onDelete: Cascade)

  @@map("company_verifications")
}

// ==========================================
// VAGAS & CANDIDATURAS
// ==========================================
model Job {
  id          String    @id @default(uuid())
  companyId   String
  title       String
  description String
  status      JobStatus @default(DRAFT)
  createdAt   DateTime  @default(now())
  updatedAt   DateTime  @updatedAt

  company      CompanyProfile @relation(fields: [companyId], references: [id], onDelete: Cascade)
  applications Application[]
  skills       JobSkill[]

  @@map("jobs")
}

model Application {
  id             String            @id @default(uuid())
  jobId          String
  candidateId    String
  status         ApplicationStatus @default(PENDING)
  resumeSnapshot Json              // Imutabilidade do currículo no envio
  createdAt      DateTime          @default(now())

  job       Job              @relation(fields: [jobId], references: [id], onDelete: Cascade)
  candidate CandidateProfile @relation(fields: [candidateId], references: [id], onDelete: Cascade)

  @@unique([jobId, candidateId]) // Previne candidaturas duplicadas
  @@map("applications")
}

model Resume {
  id          String   @id @default(uuid())
  candidateId String
  title       String
  isPrimary   Boolean  @default(false)
  createdAt   DateTime @default(now())

  candidate CandidateProfile @relation(fields: [candidateId], references: [id], onDelete: Cascade)

  @@map("resumes")
}

model Evidence {
  id          String         @id @default(uuid())
  candidateId String
  source      EvidenceSource
  type        String
  title       String
  url         String
  metadata    Json           // Dados flexíveis JSONB (stars, repositórios)
  createdAt   DateTime       @default(now())

  candidate CandidateProfile @relation(fields: [candidateId], references: [id], onDelete: Cascade)

  @@map("evidences")
}

// ==========================================
// HABILIDADES & MATCHING (JUNÇÃO N:M)
// ==========================================
model Skill {
  id   String @id @default(uuid())
  name String @unique

  candidateSkills CandidateSkill[]
  jobSkills       JobSkill[]

  @@map("skills")
}

model CandidateSkill {
  candidateId String
  skillId     String
  yearsOfExp  Int    @default(0)

  candidate CandidateProfile @relation(fields: [candidateId], references: [id], onDelete: Cascade)
  skill     Skill            @relation(fields: [skillId], references: [id], onDelete: Cascade)

  @@id([candidateId, skillId]) // Chave Primária Composta
  @@map("candidate_skills")
}

model JobSkill {
  jobId       String
  skillId     String
  isMandatory Boolean @default(true)

  job   Job   @relation(fields: [jobId], references: [id], onDelete: Cascade)
  skill Skill @relation(fields: [skillId], references: [id], onDelete: Cascade)

  @@id([jobId, skillId]) // Chave Primária Composta
  @@map("job_skills")
}
Milestone 0 • Infraestrutura & Server Engine

Fundação Local, Fastify Adapter e IoC Container

🧠 Fundamentos da Etapa: O que ocorre por baixo dos panos?

No Express tradicional, o fluxo é imperativo. No NestJS, adotamos **Inversão de Controle (IoC)** e **Injeção de Dependência (DI)**. O IoC Container gerencia o ciclo de vida e a instanciação das classes marcadas com @Injectable().

TypeScript Decorator Metadata: Ao habilitar emitDecoratorMetadata no tsconfig.json, o compilador do TypeScript converte a assinatura do construtor em chamadas Reflect.metadata("design:paramtypes", [...]) no JS transpilado. Em runtime, o NestJS lê esse mapa para resolver automaticamente as instâncias requeridas.

🛠️ Sua Vez: Desafio Prático Hands-on

Compare a sua implementação do Healthcheck E2E com este modelo de referência:

// test/e2e/health.e2e-spec.ts 
describe('Healthcheck (E2E)', () => {
  it('GET /health deve responder 200 OK e status do banco', async () => {
    const response = await app.inject({
      method: 'GET',
      url: '/health',
    });

    expect(response.statusCode).toBe(200);
    expect(JSON.parse(response.payload)).toEqual({
      status: 'ok',
      database: 'up',
    });
  });
});
Milestone 1 • Autenticação & Segurança

Auth, Users, ValidationPipe e JWT Refresh Tokens

🧠 Fundamentos: DTOs em TypeScript e Ciclo de Vida do Guard

Por que usar Classes e não Interfaces para DTOs? Em TypeScript, interface e type sofrem Type Erasure durante a transpilação e desaparecem do JS gerado. Se um DTO for uma interface, o NestJS e o class-validator não conseguirão inspecionar metadados em tempo de execução. DTOs devem ser Classes.

Guards (CanActivate): O Guard executa após os Middlewares e antes dos Interceptors e Pipes. Ele recebe o ExecutionContext, permitindo tomar decisões de acesso com base na rota invocada.

🛠️ Sua Vez: Desafio Prático Hands-on

Exemplo de DTO de Login com validação e Guard JWT:

// src/modules/auth/dto/login.dto.ts 
import { IsEmail, IsNotEmpty, MinLength } from 'class-validator';

export class LoginDto {
  @IsEmail({}, { message: 'Formato de e-mail inválido' })
  @IsNotEmpty()
  email: string;

  @MinLength(6, { message: 'Senha deve conter ao menos 6 caracteres' })
  password: string;
}
Milestone 2 • Perfis & Validação Empresarial

Candidates, Companies e Custom Param Decorators

🧠 Custom Param Decorators & Relacionamentos 1:1 no Prisma

Custom Param Decorator: Permite extrair dados limpos da requisição HTTP sem acoplar o Controller ao objeto de transporte (req.user). Criado com createParamDecorator().

Relacionamento 1:1 no Prisma: Exige obrigatoriamente a restrição @unique no campo de chave estrangeira do modelo filho.

🛠️ Sua Vez: Desafio Prático Hands-on

// src/common/decorators/current-user.decorator.ts 
import { createParamDecorator, ExecutionContext } from '@nestjs/common';

export const CurrentUser = createParamDecorator(
  (data: string | undefined, ctx: ExecutionContext) => {
    const request = ctx.switchToHttp().getRequest();
    const user = request.user;
    return data ? user?.[data] : user;
  },
);
Milestone 3 • Vagas & Candidaturas

Interceptors, Snapshots & Transações Interativas

🧠 Programação Orientada a Aspectos (AOP) e Padrão Snapshot

Interceptors com RxJS: Permitem envelopar o retorno da rota antes de enviar a resposta HTTP ao cliente, padronizando formatos JSON.

Padrão Snapshot: Grava uma cópia imutável (JSON) do currículo na tabela de candidaturas no instante do envio, prevenindo que alterações futuras afetem a candidatura realizada.

🛠️ Sua Vez: Desafio Prático Hands-on

// Transação Interativa de Candidatura 
await this.prisma.$transaction(async (tx) => {
  const resume = await tx.resume.findUnique({ where: { id: resumeId } });
  return tx.application.create({
    data: {
      jobId,
      candidateId,
      resumeSnapshot: JSON.stringify(resume),
    },
  });
});
Milestone 4 • Processamento Assíncrono

Redis, BullMQ e Worker Process Isolado

🧠 Arquitetura de Filas e Nest Standalone Context

Ao desacoplar tarefas pesadas (envio de e-mails/sincronizações), a API HTTP apenas insere a tarefa no Redis e responde HTTP 202 Accepted. Um processo Worker separado consome a fila em background sem bloquear o Event Loop da API.

🛠️ Sua Vez: Desafio Prático Hands-on

// src/worker.ts 
import { NestFactory } from '@nestjs/core';
import { WorkerModule } from './worker.module';

async function bootstrap() {
  const app = await NestFactory.createApplicationContext(WorkerModule);
  app.enableShutdownHooks();
  console.log('Worker isolado do BullMQ executando...');
}
bootstrap();

Milestone 6 • Inteligência & Recomendações

Algoritmo de Matching e Caching com Redis

🧠 N:M Explícito e Padrão Cache-Aside

Cache-Aside: A requisição busca primeiro a pontuação no Redis. Se não existir (Cache Miss), o cálculo é processado, salvo no Redis com TTL de 30 minutos e então retornado ao cliente.

🛠️ Sua Vez: Desafio Prático Hands-on

// Exemplo de Cache-Aside no MatchingService 
async getScore(jobId: string, candidateId: string) {
  const key = `match:${jobId}:${candidateId}`;
  const cached = await this.redis.get(key);
  if (cached) return JSON.parse(cached);

  const score = await this.calculateLogic(jobId, candidateId);
  await this.redis.set(key, JSON.stringify(score), 'EX', 1800);
  return score;
}
